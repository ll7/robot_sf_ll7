"""Scan every JSON/YAML/CSV under a resolved-plan dir for seed-named values in 111..140."""
import json, sys, re
from pathlib import Path
root = Path(sys.argv[1])
stats_rng = []; hits = []; seeds_seen = set(); files = 0
def walk(o, path=""):
    if isinstance(o, dict):
        for k, v in o.items(): yield from walk(v, f"{path}.{k}")
    elif isinstance(o, list):
        for i, v in enumerate(o): yield from walk(v, f"{path}[{i}]")
    else: yield path, o
for f in sorted(root.rglob("*")):
    if not f.is_file(): continue
    if f.suffix == ".json":
        files += 1
        try: data = json.loads(f.read_text())
        except Exception: continue
        for p, v in walk(data):
            leaf = p.lower()
            if "seed" in leaf and isinstance(v, int) and not isinstance(v, bool):
                if re.search(r"seeds?(\[\d+\])?$", leaf) or leaf.endswith("seed"):
                    seeds_seen.add(v)
                    # statistics RNG seeds (numpy default_rng for bootstrap / SNQI weight
                    # calibration) never drive a planner or environment step; report them apart.
                    if 111 <= v <= 140 or 50000 <= v <= 59999:
                        if leaf.endswith((".calibration_seed", ".bootstrap_seed")):
                            stats_rng.append((str(f.relative_to(root)), p, v))
                        else:
                            hits.append((str(f.relative_to(root)), p, v))
    elif f.suffix == ".jsonl":
        files += 1
        for ln, line in enumerate(f.read_text(errors="ignore").splitlines(), 1):
            if not line.strip(): continue
            for p, v in walk(json.loads(line)):
                leaf = p.lower()
                if "seed" in leaf and isinstance(v, int) and not isinstance(v, bool) and (111 <= v <= 140 or 50000 <= v <= 59999):
                    hits.append((str(f.relative_to(root)), ln, p, v))
    elif f.suffix in (".csv", ".md", ".yaml"):
        files += 1
        for ln, line in enumerate(f.read_text(errors="ignore").splitlines(), 1):
            if re.search(r"seed", line, re.I) and not re.search(r"(calibration|bootstrap)_seed", line) and re.search(r"\b(11[1-9]|12\d|13\d|140|5\d{4})\b", line):
                hits.append((str(f.relative_to(root)), ln, line[:160]))
print("files scanned:", files)
print("seed-keyed integer values seen:", sorted(seeds_seen))
print("statistics-RNG seeds in band (not episode seeds):", sorted({x[1] for x in stats_rng}))
print("held-out-band hits:", len(hits))
for h in hits[:30]: print("  ", h)
sys.exit(3 if hits else 0)
